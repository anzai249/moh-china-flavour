moh_chengsheng_gengzhang_fiscal_status = {
	type = country
	random_valid = no
	text = {
		trigger = {
			OR = {
				AND = {
					NOT = { has_variable = moh_chengsheng_gengzhang_settled }
					moh_chengsheng_gengzhang_fiscal_reform = yes
				}
				AND = {
					has_variable = moh_chengsheng_gengzhang_settled
					has_variable = moh_chengsheng_gengzhang_fiscal_completed
				}
			}
		}
		localization_key = je_moh_chengsheng_gengzhang_fiscal_done
	}
	text = { localization_key = je_moh_chengsheng_gengzhang_fiscal_pending }
}
