moh_resume_chengsheng_gengzhang = {
	is_shown = {
		c:CHI ?= this
		has_variable = moh_chengsheng_gengzhang_deferred
		NOT = { has_variable = moh_chengsheng_gengzhang_started }
		NOT = { has_variable = moh_chengsheng_gengzhang_outcome }
		NOT = { has_journal_entry = je_moh_chengsheng_gengzhang }
	}
	possible = { moh_chengsheng_gengzhang_eligible = yes }
	when_taken = {
		custom_tooltip = {
			text = moh_chengsheng_gengzhang_start_tt
			set_variable = moh_chengsheng_gengzhang_available
			remove_variable = moh_chengsheng_gengzhang_deferred
			add_journal_entry = { type = je_moh_chengsheng_gengzhang }
		}
	}
	ai_chance = { value = 10 }
}

moh_qing_shanhai_pass = {
	is_shown = {
		exists = c:CHI
		c:CHI = ROOT
		any_scope_state = {
			has_state_trait = state_trait_dragon_rise_land
		}
	}
	possible = {
		exists = c:CHI
		c:CHI = ROOT
		any_scope_state = {
			has_state_trait = state_trait_dragon_rise_land
		}
		has_technology_researched = nationalism
		OR = {
			any_state = {
				OR = {
					state_region = s:STATE_AMUR
					state_region = s:STATE_OUTER_MANCHURIA
				}
				NOT = { owner = c:CHI }
			}
			any_diplomatic_play = {
				target_is = root
				initiator_is = c:RUS
			}
		}
	}
	when_taken = {
		every_scope_state = {
			limit = {
				has_state_trait = state_trait_dragon_rise_land
			}
			state_region = {
				remove_state_trait = state_trait_dragon_rise_land
			}
		}
	}
	ai_chance = {
		value = 10
	}
}
