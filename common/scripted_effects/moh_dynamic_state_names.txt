# Extend vanilla routing using the geographic regions that preserve the old boundaries.
INJECT:assign_state_hub_dynamic_names = {
	if = {
		limit = {
			is_in_geographic_region = geographic_region_west_siberia_old
		}
		assign_state_name_region_west_siberia = yes
	}
	if = {
		limit = {
			is_in_geographic_region = geographic_region_central_asia_old
		}
		assign_state_name_region_central_asia = yes
	}
    if = {
        limit = {
            is_in_geographic_region = geographic_region_south_china_old
        }
        assign_state_name_region_south_china = yes
    }
}

## South China

# Vanilla already dispatches South China; append the mod rule to that effect.
INJECT_OR_CREATE:assign_state_name_region_south_china = {
    if = {
        limit = {
            state_region = {
                s:STATE_SHAOZHOU ?= this
            }
        }
        STATE_SHAOZHOU_state_name_assign = yes
    }
}

STATE_SHAOZHOU_state_name_assign = {
    if = {
        limit = {
            owner = {
                NOT = {
                    country_has_primary_culture = cu:han
                }
            }
        }
		reset_state_name = yes
        reset_hub_names = yes
    }
}
