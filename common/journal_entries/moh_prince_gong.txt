					var:prince_gong_progress >= 75

			triggered_desc = {
				desc = je_prince_gong_status_low

		}

					any_scope_character = {

				}
			}
		}
	}

		conclusion_desc = je_standard_failure_conclusion_desc

			custom_tooltip = {
