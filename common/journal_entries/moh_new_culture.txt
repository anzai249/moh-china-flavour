je_new_culture = {
	icon = "gfx/interface/icons/event_icons/new_culture.dds"
	group = je_group_historical_content
	is_shown_when_inactive = {
		exists = c:CHI
		c:CHI = root
		NOT = {
			has_law = law_type:law_monarchy
		}
		NOT = {
			has_variable = new_culture
		}
		always = no
	}
	possible = {
		exists = c:CHI
		c:CHI ?= this
		has_law = law_type:law_monarchy
		OR = {
			has_law = law_type:law_right_of_assembly
			has_law = law_type:law_protected_speech
		}
		always = no
	}
	immediate = {
		set_variable = new_culture
	}
	scripted_progress_bar = new_culture_communist_progress_bar
	scripted_progress_bar = new_culture_united_provinces_progress_bar
	scripted_progress_bar = new_culture_monarchy_progress_bar
	scripted_progress_bar = new_culture_anarchist_progress_bar
	scripted_progress_bar = new_culture_presidential_autocracy_progress_bar
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = new_culture_complete_tt
				always = no
			}
		}
	}
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			always = no
			OR = {
				has_law = law_type:law_council_republic
				has_law = law_type:law_monarchy
				has_law = law_type:law_anarchy
				AND = {
					has_law = law_type:law_presidential_republic
					has_law = law_type:law_autocracy
				}
			}
		}
	}
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_modifier = {
				name = modifier_conservatism_grow
				years = 5
			}
		}
	}
	timeout = 1825
}
