je_self_strengthening = {
	activation_description = {
		always = yes
	}

    group = je_group_historical_content
    icon = "gfx/interface/icons/invention_icons/manufacturies.dds"
    modifiers_while_active = {
        modifier_self_strengthening
    }
    is_shown_in_lobby = {
        c:CHI ?= THIS
    }
    immediate = {
        set_variable = {
            name = self_strengthening_progress
            value = 0
        }
        set_variable = {
            name = self_strengthening_gdp
            value = gdp
        }
        set_variable = {
            name = self_strengthening_lost_wars
            value = 0
        }
        add_journal_entry = {
            type = je_self_strengthening_military
        }
        add_journal_entry = {
            type = je_self_strengthening_industry
        }
        add_journal_entry = {
            type = je_self_strengthening_education
        }
        add_journal_entry = {
            type = je_self_strengthening_diplomacy
        }
    }
    current_value = {
        value = var:self_strengthening_progress
    }
    goal_add_value = {
        add = 3
    }
    progressbar = yes
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = moh_je_self_strengthening_completed_desc
        trigger = {
            custom_tooltip = {
                text = je_self_strengthening_main_complete_tt
                var:self_strengthening_progress >= 3
            }
        }
        effect = {
            custom_tooltip = {
                text = je_self_strengthening_on_complete_tt
                if = {
                    limit = {
                        has_journal_entry = je_earn_recognition
                    }
                    je:je_earn_recognition ?= {
                        add_progress = { value = 20 name = earn_recognition_progress_bar }
                    }
                }
            }
            add_modifier = {
                name = modifier_self_strengthening_success
                years = 10
                is_decaying = yes
            }
            add_modifier = {
                name = modifier_self_strengthening_success_company
            }
            set_variable = self_strengthening_end
        }
    }
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = moh_je_self_strengthening_failed_desc
        trigger = {
            OR = {
                NOT = {
                    has_state_in_state_region = STATE_FORMOSA
                    c:DAI = {
                        is_subject_of = c:CHI
                    }
                }
                custom_tooltip = {
                    text = je_self_strengthening_fail_tt
                    var:self_strengthening_lost_wars >= 2
                }
            }
        }
        effect = {
            set_variable = self_strengthening_end
        }
    }
    outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        flags = { timeout }
        conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
        effect = {
            set_variable = self_strengthening_end
        }
    }
    weight = 100
    should_be_pinned_by_default_uninvolved_or_context = yes
    timeout = 14600
}

je_self_strengthening_military = {
	activation_description = {
		always = yes
	}

    group = je_group_historical_content
    icon = "gfx/interface/icons/invention_icons/battleship_tech.dds"
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = je_self_strengthening.tt.complete.world
                any_country = {
                    naval_power_projection > root.naval_power_projection
                    count < 3
                }
            }
            custom_tooltip = {
                text = je_self_strengthening.tt.complete.asia
                any_country = {
                    naval_power_projection > root.naval_power_projection
                    count < 1
                    OR = {
                        is_in_geographic_region = geographic_region_india
                        is_in_geographic_region = geographic_region_east_asia
                        is_in_geographic_region = geographic_region_south_east_asia
                        is_in_geographic_region = geographic_region_central_asia
                        is_in_geographic_region = geographic_region_middle_east
                    }
                }
            }
        }
        effect = {
            custom_tooltip = {
                text = je_self_strengthening_complete_tt
                change_variable = {
                    name = self_strengthening_progress
                    add = 1
                }
            }
        }
    }
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            NOT = {
                has_journal_entry = je_self_strengthening
            }
        }
    }
    on_yearly_pulse = {
        random_events = {
            10 = 0
            2 = self_strengthening.3
        }
    }
    weight = 100
    should_be_pinned_by_default_uninvolved_or_context = yes
}

            }
